Instrument one scope, tool call, and LLM call with NeMo Relay, then verify the
result in a local ATOF event file.

## Prerequisites

- Node.js 24 or newer
- npm

## 1. Install

Create a project and install the pinned published package. In an existing
Node.js project, skip `npm init`.

{/* quickstart-block {"id":"install","role":"run","cwd":".","shells":["bash","powershell"],"session":"main","timeout_seconds":600} */}
```bash
mkdir relay-node-quickstart
cd relay-node-quickstart
npm init -y
npm install --save-exact nemo-relay-node@0.10.0
```

## 2. Configure

Save the following configuration as `plugins.toml` in your project directory:

{/* quickstart-block {"id":"plugins","role":"file","cwd":"relay-node-quickstart","path":"plugins.toml"} */}
```toml
version = 1

[policy]
unknown_component = "error"
unknown_field = "error"
unsupported_value = "error"

[[components]]
kind = "observability"
enabled = true

[components.config]
version = 4
enable_full_payloads = false

[components.config.atof]
enabled = true

[[components.config.atof.sinks]]
type = "file"
output_directory = "logs"
filename = "events.jsonl"
mode = "overwrite"
```

ATOF output can contain sensitive prompts, responses, tool arguments, and tool
results. Protect `logs/events.jsonl` even when full payload export is disabled.

## 3. Run

Save this file as `quickstart.js` beside `plugins.toml`:

{/* quickstart-block {"id":"program","role":"file","cwd":"relay-node-quickstart","path":"quickstart.js"} */}
```js
const fs = require("node:fs");
const relay = require("nemo-relay-node");
const plugin = require("nemo-relay-node/plugin");

async function main() {
  const activation = await plugin.initialize(plugin.defaultConfig(), "plugins.toml");

  try {
    await relay.withScope("demo-agent", relay.ScopeType.Agent, async (handle) => {
      relay.event("initialized", handle, { binding: "node" }, null);

      const toolResult = await relay.toolCallExecute(
        "emit_marker",
        { marker: "RELAY_QUICKSTART_TOOL_OK" },
        (args) => ({ result: { marker: args.marker } }),
        handle,
        null,
        null,
        null,
      );

      const llmResult = await relay.llmCallExecute(
        "demo-provider",
        new relay.LlmRequest(
          {},
          { messages: [{ role: "user", content: "RELAY_QUICKSTART_RESPONSE_OK" }] },
        ),
        () => ({ content: "RELAY_QUICKSTART_RESPONSE_OK" }),
        handle,
        null,
        null,
        null,
        null,
      );

      console.log(toolResult.result.marker);
      console.log(llmResult.content);
    });
  } finally {
    try {
      await relay.flushSubscribers();
    } finally {
      await activation.close();
    }
  }

  const eventText = fs.readFileSync("logs/events.jsonl", "utf8").trim();
  if (!eventText) throw new Error("Relay event file is empty");
  const events = eventText.split(/\r?\n/).map((line) => JSON.parse(line));
  if (events.some((event) => !event || event.atof_version !== "0.1" ||
      !["scope", "mark"].includes(event.kind))) {
    throw new Error("Expected ATOF 0.1 records");
  }
  const starts = new Map(events.filter((event) =>
    event.kind === "scope" && event.scope_category === "start",
  ).map((event) => [event.uuid, event]));
  const closed = events.filter((event) => {
    const start = starts.get(event.uuid);
    return event.kind === "scope" && event.scope_category === "end" && start &&
      ["name", "category", "parent_uuid"].every((key) => event[key] === start[key]);
  });
  if (!["agent", "tool", "llm"].every((category) =>
      closed.some((event) => event.category === category))) {
    throw new Error("Expected paired agent, tool, and LLM lifecycle records");
  }
  console.log("RELAY_QUICKSTART_EVENTS_OK");
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

Run it from the project directory so the relative output path resolves there:

{/* quickstart-block {"id":"run","role":"run","cwd":"relay-node-quickstart","shells":["bash","powershell"],"session":"main","timeout_seconds":180} */}
```bash
node quickstart.js
```

## 4. Verify

The program parses every JSONL record and checks for paired agent, tool, and
LLM lifecycle records before printing, in order:

{/* quickstart-block {"id":"expected","role":"expected","cwd":"relay-node-quickstart","from":"run","match":"exact-lines"} */}
```text
RELAY_QUICKSTART_TOOL_OK
RELAY_QUICKSTART_RESPONSE_OK
RELAY_QUICKSTART_EVENTS_OK
```

The event file contains scope, tool, and LLM records in ATOF JSONL format.

Instrument one scope, tool call, and LLM call with NeMo Relay, then verify the
result in a local ATOF event file.

## Install NeMo Relay

From your Node.js project, install the published package:

```bash
npm install nemo-relay-node@0.10.0
```

When working from a NeMo Relay source checkout instead, run
`npm install --ignore-scripts` followed by
`npm run build --workspace=nemo-relay-node`.

## Enable Observability

Save the following configuration as `plugins.toml` in your project directory:

```toml
version = 1

[policy]
unknown_component = "error"
unknown_field = "error"
unsupported_value = "error"

[[components]]
kind = "observability"
enabled = true

[components.config]
version = 4
enable_full_payloads = false

[components.config.atof]
enabled = true

[[components.config.atof.sinks]]
type = "file"
output_directory = "logs"
filename = "events.jsonl"
mode = "overwrite"
```

ATOF output can contain sensitive prompts, responses, tool arguments, and tool
results. Protect `logs/events.jsonl` even when full payload export is disabled.

## Run an Instrumented Workflow

Save this file as `quickstart.js` beside `plugins.toml`:

```js
const relay = require("nemo-relay-node");
const plugin = require("nemo-relay-node/plugin");

async function main() {
  const activation = await plugin.initialize(plugin.defaultConfig(), "plugins.toml");

  try {
    await relay.withScope("demo-agent", relay.ScopeType.Agent, async (handle) => {
      relay.event("initialized", handle, { binding: "node" }, null);

      const toolResult = await relay.toolCallExecute(
        "search",
        { query: "hello" },
        (args) => ({ result: { echo: args.query } }),
        handle,
        null,
        null,
        null,
      );

      const llmResult = await relay.llmCallExecute(
        "demo-provider",
        new relay.LlmRequest({}, { messages: [{ role: "user", content: "hi" }] }),
        (request) => ({ ok: true, messages: request.content.messages }),
        handle,
        null,
        null,
        null,
        null,
      );

      console.log(toolResult.result);
      console.log(llmResult);
    });

    await relay.flushSubscribers();
  } finally {
    await activation.close();
  }
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

Run it from the project directory so the relative output path resolves there:

```bash
node quickstart.js
```

## Verify the Events

The program prints the synthetic tool and LLM results. Confirm that Relay also
wrote lifecycle events:

```bash
test -s logs/events.jsonl
tail -n 1 logs/events.jsonl | python3 -m json.tool
```

The file contains scope, tool, and LLM records in ATOF JSONL format.
